Signal class implementation for Roblox Luau
- Signal.new() creates a new signal
- :Connect(fn) returns a connection with :Disconnect()
- :Fire(...) calls all connected functions with the same table reference if table passed

		self._active = false
		for i, c in ipairs(self._signal._connections) do
			if c == self then
				table.remove(self._signal._connections, i)

		end
	end

function Signal:Fire(...)
	local args = {...}
	-- Check if first argument is a table and we should pass same table reference
	local useSameTable = false
	if #args > 0 and type(args[1]) == "table" then
		useSameTable = true
	end
	
	for _, conn in ipairs(self._connections) do
		if conn._active then
			if useSameTable and type(args[1]) == "table" then
				-- Pass the same table reference
				conn.fn(table.unpack(args))
			else
				conn.fn(table.unpack(args))
			end
		end
	end
end

function Signal:Once(fn)

function Signal:Fire(...)
