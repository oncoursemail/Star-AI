local PriorityQueue = {}
PriorityQueue.__index = PriorityQueue

function PriorityQueue.new(less: ((any, any) -> boolean)?)
	local self = setmetatable({}, PriorityQueue)
	self._heap = {}
	self._size = 0
	if less then
		self._less = less
	else
		self._less = function(a, b)
			return a < b
		end
	end
	return self
end

function PriorityQueue:_siftUp(index: number)
	while index > 1 do
		local parent = math.floor(index / 2)
		if self._less(self._heap[index], self._heap[parent]) then
			self._heap[index], self._heap[parent] = self._heap[parent], self._heap[index]
			index = parent
		else
			break
		end
	end
end

function PriorityQueue:_siftDown(index: number)
	local size = self._size
	while true do
		local left = 2 * index
		local right = 2 * index + 1
		local smallest = index

		if left <= size and self._less(self._heap[left], self._heap[smallest]) then
			smallest = left
		end

		if right <= size and self._less(self._heap[right], self._heap[smallest]) then
			smallest = right
		end

		if smallest ~= index then
			self._heap[index], self._heap[smallest] = self._heap[smallest], self._heap[index]
			index = smallest
		else
			break
		end
	end
end

function PriorityQueue:push(value: any)
	self._size = self._size + 1
	self._heap[self._size] = value
	self:_siftUp(self._size)
end

function PriorityQueue:pop(): any?
	if self._size == 0 then
		return nil
	end
	local top = self._heap[1]
	self._heap[1] = self._heap[self._size]
	self._heap[self._size] = nil
	self._size = self._size - 1
	if self._size > 0 then
		self:_siftDown(1)
	end
	return top
end

function PriorityQueue:peek(): any?
	if self._size == 0 then
		return nil
	end
	return self._heap[1]
end

function PriorityQueue:size(): number
	return self._size
end

return PriorityQueue
