local Inventory = {}
Inventory.__index = Inventory

function Inventory.new(maxStack, slots)
	local self = setmetatable({}, Inventory)
	self.maxStack = maxStack
	self.slots = slots
	self.items = {}
	for i = 1, slots do
		self.items[i] = { item = nil, count = 0 }
	end
	return self
end

function Inventory:add(item, n)
	if n <= 0 then return 0 end
	local remaining = n
	local added = 0

	-- First pass: fill existing stacks of the same item
	for i = 1, self.slots do
		local slot = self.items[i]
		if slot.item == item and slot.count < self.maxStack then
			local space = self.maxStack - slot.count
			local toAdd = math.min(space, remaining)
			slot.count = slot.count + toAdd
			remaining = remaining - toAdd
			added = added + toAdd
			if remaining == 0 then break end
		end
	end

	-- Second pass: fill empty slots with the item
	if remaining > 0 then
		for i = 1, self.slots do
			local slot = self.items[i]
			if slot.item == nil then
				local toAdd = math.min(self.maxStack, remaining)
				slot.item = item
				slot.count = toAdd
				remaining = remaining - toAdd
				added = added + toAdd
				if remaining == 0 then break end
			end
		end
	end

	return added
end
