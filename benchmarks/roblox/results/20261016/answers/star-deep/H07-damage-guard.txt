local DamageGuard = {}
DamageGuard.__index = DamageGuard

function DamageGuard.new()
	local self = setmetatable({}, DamageGuard)
	self._lastAllowed = {}
	return self
end

local function getHumanoid(model)
	if model and model:IsA("Model") then
		local humanoid = model:FindFirstChildOfClass("Humanoid")
		return humanoid
	end
	return nil
end

local function getPrimaryPart(model)
	if model and model:IsA("Model") then
		local pp = model.PrimaryPart
		if pp and pp:IsA("BasePart") then
			return pp
		end
		local hrp = model:FindFirstChild("HumanoidRootPart")
		if hrp and hrp:IsA("BasePart") then
			return hrp
		end
	end
	return nil
end

function DamageGuard:allow(player, target, amount, now)
	if typeof(player) ~= "Instance" or not player:IsA("Player") then
		return false
	end
	if typeof(now) ~= "number" or not (now == now) or now == math.huge or now == -math.huge then
		return false
	end
	if typeof(amount) ~= "number" or not (amount == amount) or amount == math.huge or amount == -math.huge then
		return false
	end
	if not (amount > 0 and amount <= 25) then
		return false
	end
	if typeof(target) ~= "Instance" or not target:IsA("Model") then
		return false
	end
	local humanoid = getHumanoid(target)
	if not humanoid or typeof(humanoid.Health) ~= "number" or humanoid.Health <= 0 then
		return false
	end
	local character = player.Character
	if character == target then
		return false
	end
	local playerHrp = nil
	if character and character:IsA("Model") then
		local pp = character.PrimaryPart
		if pp and pp:IsA("BasePart") then
			playerHrp = pp
		else
			local hrp = character:FindFirstChild("HumanoidRootPart")
			if hrp and hrp:IsA("BasePart") then
				playerHrp = hrp
			end
		end
	end
	if not playerHrp then
		return false
	end
	local targetPart = getPrimaryPart(target)
	if not targetPart then
		return false
	end
	local dist = (targetPart.Position - playerHrp.Position).Magnitude
	if not (dist == dist) or dist > 15 then
		return false
	end
	local last = self._lastAllowed[player]
	if last ~= nil and (now - last) < 0.5 then
		return false
	end
	self._lastAllowed[player] = now
	return true
end

return DamageGuard
