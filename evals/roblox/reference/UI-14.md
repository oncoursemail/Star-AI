local function createNameplate(parent: Instance, displayName: string): TextLabel
	local label = Instance.new("TextLabel")
	label.Size = UDim2.fromOffset(200, 40)
	label.BackgroundTransparency = 1
	label.Text = displayName
	label.TextScaled = true
	label.TextWrapped = false
	label.ClipsDescendants = true
	local limits = Instance.new("UITextSizeConstraint")
	limits.MinTextSize = 10
	limits.MaxTextSize = 24
	limits.Parent = label
	label.Parent = parent
	return label
end
return createNameplate
