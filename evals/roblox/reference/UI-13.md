local Players = game:GetService("Players")
local TweenService = game:GetService("TweenService")

local button = Players.LocalPlayer:WaitForChild("PlayerGui"):WaitForChild("MainMenu"):WaitForChild("PlayButton")
local info = TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
local baseSize = button.Size
local current: Tween? = nil

local function scaleTo(factor: number)
	if current then
		current:Cancel()
	end
	local target = UDim2.new(baseSize.X.Scale * factor, baseSize.X.Offset * factor, baseSize.Y.Scale * factor, baseSize.Y.Offset * factor)
	current = TweenService:Create(button, info, { Size = target })
	current:Play()
end

local connections = {
	button.MouseEnter:Connect(function() scaleTo(1.05) end),
	button.MouseLeave:Connect(function() scaleTo(1) end),
	button.SelectionGained:Connect(function() scaleTo(1.05) end),
	button.SelectionLost:Connect(function() scaleTo(1) end),
	button.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			scaleTo(0.95)
		end
	end),
	button.InputEnded:Connect(function() scaleTo(1.05) end),
}

button.Destroying:Connect(function()
	if current then
		current:Cancel()
	end
	for _, c in connections do
		c:Disconnect()
	end
end)
