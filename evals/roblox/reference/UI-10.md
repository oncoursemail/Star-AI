local Players = game:GetService("Players")
local shopGui = Players.LocalPlayer:WaitForChild("PlayerGui"):WaitForChild("ShopGui")

local list = Instance.new("ScrollingFrame")
list.Size = UDim2.fromScale(0.5, 0.7)
list.Position = UDim2.fromScale(0.25, 0.15)
list.CanvasSize = UDim2.new()
list.AutomaticCanvasSize = Enum.AutomaticSize.Y
list.ScrollingDirection = Enum.ScrollingDirection.Y
list.Parent = shopGui

local layout = Instance.new("UIListLayout")
layout.SortOrder = Enum.SortOrder.LayoutOrder
layout.Padding = UDim.new(0, 6)
layout.Parent = list

local padding = Instance.new("UIPadding")
padding.PaddingLeft = UDim.new(0, 8)
padding.PaddingRight = UDim.new(0, 8)
padding.Parent = list

for i = 1, 30 do
	local row = Instance.new("Frame")
	row.LayoutOrder = i
	row.Size = UDim2.new(1, 0, 0, 40)
	row.Parent = list

	local name = Instance.new("TextLabel")
	name.Size = UDim2.fromScale(0.7, 1)
	name.TextXAlignment = Enum.TextXAlignment.Left
	name.Text = "Item " .. i
	name.Parent = row

	local price = Instance.new("TextLabel")
	price.AnchorPoint = Vector2.new(1, 0)
	price.Position = UDim2.fromScale(1, 0)
	price.Size = UDim2.fromScale(0.3, 1)
	price.TextXAlignment = Enum.TextXAlignment.Right
	price.Text = tostring(i * 10)
	price.Parent = row
end
