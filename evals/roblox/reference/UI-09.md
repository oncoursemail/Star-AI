A LocalScript that builds the HUD with a grid layout, Scale sizes and a square constraint.

```luau
local Players = game:GetService("Players")

local gui = Instance.new("ScreenGui")
gui.Name = "InventoryHud"
gui.ResetOnSpawn = false
gui.Parent = Players.LocalPlayer:WaitForChild("PlayerGui")

local panel = Instance.new("Frame")
panel.AnchorPoint = Vector2.new(0.5, 0.5)
panel.Position = UDim2.fromScale(0.5, 0.5)
panel.Size = UDim2.fromScale(0.6, 0.7)
panel.Parent = gui

local panelRatio = Instance.new("UIAspectRatioConstraint")
panelRatio.AspectRatio = 1.2
panelRatio.Parent = panel

local title = Instance.new("TextLabel")
title.Size = UDim2.fromScale(0.7, 0.12)
title.BackgroundTransparency = 1
title.Text = "Inventory"
title.TextScaled = true
title.Parent = panel

local close = Instance.new("TextButton")
close.AnchorPoint = Vector2.new(1, 0)
close.Position = UDim2.fromScale(1, 0)
close.Size = UDim2.fromScale(0.12, 0.12)
close.Text = "X"
close.TextScaled = true
close.Parent = panel
close.Activated:Connect(function()
	gui.Enabled = false
end)

local grid = Instance.new("Frame")
grid.Position = UDim2.fromScale(0.05, 0.16)
grid.Size = UDim2.fromScale(0.9, 0.8)
grid.BackgroundTransparency = 1
grid.Parent = panel

local layout = Instance.new("UIGridLayout")
layout.CellSize = UDim2.fromScale(0.23, 0.3)
layout.CellPadding = UDim2.fromScale(0.02, 0.03)
layout.Parent = grid

for i = 1, 12 do
	local slot = Instance.new("Frame")
	slot.Name = "Slot" .. i
	slot.LayoutOrder = i
	slot.Parent = grid
	local square = Instance.new("UIAspectRatioConstraint")
	square.AspectRatio = 1
	square.Parent = slot
end
```
